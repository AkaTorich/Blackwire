# camera_payload.adr — готовый эксплоит для игровой механики overflow
# Исходник доступен для изучения и изменения в редакторе Blackwire.
if len(sys.args) < 2:
    print("Использование: camera_payload.adr адрес [id уязвимости]")
    print("Поиск цели: probe адрес; подсказки: research адрес id")
    exit(1)
try:
    address = sys.args[1]
    selector = sys.args[2] if len(sys.args) > 2 else null
    profile = net.research(address, selector)
    if profile["kind"] != "overflow":
        print("Инструмент не подходит: нужна механика", profile["kind"])
        print("Подходящий инструмент: market buy", profile["market"])
        exit(1)
    if profile["service"] != "lab_control" or not profile["version"].startswith("1."):
        print("Camera Payload подходит только к lab_control версии 1.x.")
        exit(1)
    selector = profile["id"]
    layout = profile["layout"]
    payload = {"fingerprint": profile["fingerprint"]}
    pad = "A" * (layout["frame_words"] * layout["word_size"])
    entry_address = layout["code_base"] + layout["entry_offset"]
    payload["buffer"] = pad + "|" + str(entry_address) + "|" + layout["canary"]
    check = net.probe(address, selector, payload)
    print(check["feedback"])
    if not check["ok"]:
        exit(1)
    if profile["local"]:
        result = net.privesc(payload, selector)
    else:
        result = net.exploit(address, selector, payload)
    print(result)
    if not result["ok"]:
        exit(1)
except as error:
    print("Эксплоит не выполнен:", error)
    exit(1)
