# Автор определяет новый тип на Adder, без изменения ядра.
def describe_signal(event):
    ring = "BLACKWIRE"
    shift = (event["seed"] + event["device"]["id"]) % len(ring)
    return {"layout":{"ring":ring,"shift":shift,"suffix":event["params"].get("suffix","!")},
        "fields":["signal"],
        "hints":["Нужно переставить символы кольца сигнала.","Сдвинь ring влево на shift позиций.","Склей ring[shift:] + ring[:shift] и добавь suffix."]}

def check_signal(event):
    layout = event["profile"]["layout"]
    expected = layout["ring"][layout["shift"]:] + layout["ring"][:layout["shift"]] + layout["suffix"]
    return {"ok":event["payload"].get("signal","") == expected,"feedback":"Неверная перестановка сигнала или suffix."}
