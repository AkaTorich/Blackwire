def create(event):
    world = plugin.world()
    n = plugin.get("created", 0) + 1
    ip = f"198.18.{n // 250}.{n % 250 + 1}"
    host = f"lab-camera-{n}"
    mid = plugin.spawn("device_lab:smart_camera", {"hostname": host, "ip": ip, "up": true})
    plugin.user(mid, "operator", "lab123", false)
    plugin.write(mid, "/etc/lab/device.txt", f"Камера {n}: тестовое устройство SDK")
    plugin.vulnerability(f"single_device_{n}", {"machine": mid, "service": "lab_control", "version_prefix": "1.0", "access": "info"})
    # Узел сразу появляется на игровой карте.
    world = plugin.world()
    world["player"]["known_machines"].append(mid)
    plugin.apply({"player": world["player"]})
    plugin.set("created", n)
    print(f"Создано устройство {host}: {ip}, id={mid}")
    print(f"probe {ip} — удалённые уязвимости типа и экземпляра; локальную исследуй после входа")
    print(f"research -h 3 {ip} device_lab:camera_access — подсказки и параметры")
    print(f"exploit new ~/scripts/camera.adr {ip} device_lab:camera_access — свой код")
    print("Готовый: market buy device_lab:camera_exploit; затем exploit -f ~/scripts/camera_payload.adr -s device_lab:camera_access " + ip)
